cmake_minimum_required(VERSION 3.28)
project(SnapLite VERSION 0.1.11 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "SnapLite currently requires Windows and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(SnapLite WIN32 src/main.cpp src/image.cpp src/preview_scaler.cpp src/document.cpp src/ocr.cpp src/ocr_quality.cpp src/ocr_setup.cpp src/capture_cursor.cpp src/canvas.cpp src/text_size.cpp src/text_editor.cpp src/app.cpp src/translation.cpp src/translation_layout.cpp src/translation_panel.cpp src/translation_tests.cpp src/integration.cpp src/regression.cpp resources/SnapLite.rc)
set(SNAPLITE_OUTPUT_NAME "SnapLite" CACHE STRING "Executable name for isolated local review builds")
set(SNAPLITE_PACKAGE_NAME "SnapLite.Development" CACHE STRING "Registered package identity")
set(SNAPLITE_PUBLISHER "CN=SnapLite Local Development" CACHE STRING "Package publisher subject")
set(SNAPLITE_APPLICATION_ID "SnapLite" CACHE STRING "Package application ID")
configure_file(resources/SnapLite.manifest.in generated/SnapLite.manifest @ONLY)
set_target_properties(SnapLite PROPERTIES OUTPUT_NAME "${SNAPLITE_OUTPUT_NAME}")
target_include_directories(SnapLite PRIVATE src)
target_compile_definitions(SnapLite PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(SnapLite PRIVATE /W4 /utf-8 /permissive- /Zc:__cplusplus /EHsc)
target_link_options(SnapLite PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/SnapLite.manifest)
set_property(TARGET SnapLite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(SnapLite PRIVATE user32 gdi32 shell32 ole32 comdlg32 comctl32 d2d1 dwrite windowscodecs windowsapp dwmapi shlwapi winhttp crypt32 bcrypt)
install(TARGETS SnapLite RUNTIME DESTINATION .)
enable_testing()
add_test(NAME core COMMAND SnapLite --self-test)
add_test(NAME translation COMMAND SnapLite --translation-test)
